Introduction

The US Census Bureau publishes the Public Use Microdata Sample (PUMS) database, which provides anyone with survey records via a public API. In this project, I will be creating functions to query the PUMS API, and then using the query results to investigate how rent prices vary across different factors.

Part 1: Data Processing

Parsing Variable Values

Before I can query the data, need to know what values each variable can take. Luckily, the API provides a json file that has all of the possible values of all the variables. the get_var_values function uses GET function to get the json for a specific year. Then, the json is filtered for all of the desired variables, and the result is saved into a .rds file for each year. As an example, the values for SEX are posted for 2024.

get_var_values <- function(year) {
  url <- paste0("https://api.census.gov/data/", year,
                "/acs/acs1/pums/variables.json")
  parsed <- fromJSON(rawToChar(GET(url)$content))
  
  state_var <- if (year %in% c(2021, 2022)) "ST" else "STATE"
  vars <- c("AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP", "PWGTP",
            "FER", "HHL", "SCH", "SCHL", "SEX", state_var, "REGION", "DIVISION")
  
  lapply(parsed$variables[vars], function(v) {
    if (!is.null(v$values$item)) unlist(v$values$item) else v$values$range
  })
}

The result is a character matrix where the first row holds the column names. The helper function below turns what GET() returns into a tibble

api_to_tibble <- function(response) {
  parsed <- fromJSON(rawToChar(response$content))
  colnames(parsed) <- parsed[1, ]
  as_tibble(parsed[-1, , drop = FALSE])
}

time_midpoint <- function(labels) {
  labels <- gsub("a\\.m\\.", "AM", labels)
  labels <- gsub("p\\.m\\.", "PM", labels)
  parts <- strsplit(labels, " to ")
  start <- parse_date_time(sapply(parts, `[`, 1), "I:M p", quiet = TRUE)
  end <- parse_date_time(sapply(parts, `[`, 2), "I:M p", quiet = TRUE)
  
  wraps <- which(end < start)
  end[wraps] <- end[wraps] + days(1)
  
  hms::as_hms(start + (end - start) / 2)
}

convert_numeric <- function(x, values) {
  x <- as.numeric(x)
  if (is.character(values)) {
    na_codes <- names(values)[grepl("^N/A|suppressed", values, ignore.case = TRUE)]
    x[x %in% as.numeric(na_codes)] <- NA
  }
  x
}

convert_time <- function(x, values) {
  midpoints <- time_midpoint(values)
  midpoints[match(as.numeric(x), as.numeric(names(values)))]
}

convert_factor <- function(x, values) {
  codes <- suppressWarnings(as.numeric(names(values)))
  labels <- values[order(codes)]
  factor(values[match(suppressWarnings(as.numeric(x)), codes)],
         levels = unique(labels))
}
get_census <- function(year = 2024,num_vars = c("AGEP", "PWGTP"),cat_vars = "SEX",geography = "State",geo_values = NULL){
  if (length(year) != 1 || !(year %in% 2021:2024)) {
    stop("year must be a single value: 2021, 2022, 2023, or 2024")
  }
  
  num_options <- c("AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP", "PWGTP")
  num_vars <- unique(c("PWGTP", num_vars))
  if (!all(num_vars %in% num_options)) {
    stop("num_vars must come from: ", paste(num_options, collapse = ", "))
  }
  if (length(num_vars) < 2) {
    stop("At least one numeric variable other than PWGTP must be returned")
  }
  
  
  cat_options <- c("FER", "HHL", "SCH", "SCHL", "SEX")
  if (length(cat_vars) < 1 || !all(cat_vars %in% cat_options)) {
    stop("At least one cat_var is required, chosen from: ",
         paste(cat_options, collapse = ", "))
  }
  

  if (length(geography) != 1 ||
      !(tolower(geography) %in% c("state", "region", "division"))) {
    stop("geography must be one of 'State', 'Region', or 'Division'")
  }
  geography <- tolower(geography)
  

  var_values <- readRDS("data/pums_vars.rds")[[as.character(year)]]
  state_var <- if (year %in% c(2021, 2022)) "ST" else "STATE"
  geo_var <- switch(geography, state = state_var,
                    region = "REGION", division = "DIVISION")
  
  if (is.null(geo_values)) {
    geo_values <- switch(geography, state = "37", region = "3", division = "5")
  }
 
  valid_geo <- setdiff(names(var_values[[geo_var]]), c("72", "9", "0"))
  
  if (identical(tolower(geo_values), "all")) {
    geo_values <- "*"
  } else if (!all(geo_values %in% valid_geo)) {
    stop("Invalid geo_values. Options are: ", paste(valid_geo, collapse = ", "))
  }
  
  url <- paste0("https://api.census.gov/data/", year, "/acs/acs1/pums?get=",
                paste(c(num_vars, cat_vars), collapse = ","),
                "&for=", geography, ":", paste(geo_values, collapse = ","),
                "&key=", key)
  census_data <- api_to_tibble(GET(url))
  
  for (var in num_vars) {
    if (var %in% c("JWAP", "JWDP")) {
      census_data[[var]] <- convert_time(census_data[[var]], var_values[[var]])
    } else {
      census_data[[var]] <- convert_numeric(census_data[[var]], var_values[[var]])
    }
  }
  
  for (var in cat_vars) {
    census_data[[var]] <- convert_factor(census_data[[var]], var_values[[var]])
  }
  
  class(census_data) <- c("census", class(census_data))
  census_data}

get_multiple_years <- function(years = 2024, ...) {
  census_list <- lapply(years, function(yr) {
    get_census(year = yr, ...) |>
      mutate(year = factor(yr))
  })
  combined <- bind_rows(census_list)
  class(combined) <- unique(c("census", class(combined)))
  combined
}

summary.census <- function(object, num_vars = NULL, cat_vars = NULL, ...) {
  if (is.null(num_vars)) {
    num_vars <- setdiff(names(object)[sapply(object, is.numeric)], "PWGTP")
  }
  if (is.null(cat_vars)) {
    cat_vars <- names(object)[sapply(object, is.factor)]
  }
  
  weight <- object$PWGTP
  results <- list()
  
  for (var in num_vars) {
    keep <- !is.na(object[[var]])
    x <- object[[var]][keep]
    w <- weight[keep]
    sample_mean <- sum(x * w) / sum(w)
    sample_sd <- sqrt(sum(x^2 * w) / sum(w) - sample_mean^2)
    results[[paste0(var, "_mean")]] <- sample_mean
    results[[paste0(var, "_sd")]] <- sample_sd
  }
  
  for (var in cat_vars) {
    results[[paste0(var, "_counts")]] <- tapply(weight, object[[var]], sum,default=0)
  }
  
  results
}
plot.census <- function(x, cat_var, num_var, ...) {
  if (missing(cat_var) || missing(num_var)) {
    stop("specify one cat_var and one num_var")
  }
  ggplot(x, aes(x = get(cat_var), y = get(num_var), weight = PWGTP)) +
    geom_boxplot() +
    labs(x = cat_var, y = num_var)
}

Investigation

Is rent getting less affordable?

To test the functions that have been made for the PUMS database, we will be investigating how much of an individuals income is going to their rent in North Carolina, and how much that has changed from 2021 to 2024. We will also look at how rent differs in certain demographics. The GRPIP variable is gross rent as a percentage of income over the past 12 months. A household is considered rent-burdened at 30% of income is spent on rent.

We will be querying all four years for NC, asking for rent burden(GRPIP), age (AGEP), household language(HHL), and sex.
